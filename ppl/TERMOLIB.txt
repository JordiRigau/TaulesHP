// ===================================================================
// TERMOLIB - motor de calcul  (HP Prime G2)
//
// Motor de calcul: determinacio de regio, interpolacio lineal simple i
// doble, i cerca inversa. Els blocs de dades venen dels fitxers
// TDAT_*.hpprgm, generats per tools/gen_ppl.py (NO editar a ma).
//
// Estructura de dades (igual per a substancies pures i mescles):
//   xxST : n x 11  [T, Pl, Pv, vl, vv, ul, uv, hl, hv, sl, sv]
//   xxSP : n x 11  [P, Tl, Tv, vl, vv, ul, uv, hl, hv, sl, sv]
//   xxIX : m x 4   [P, primera_fila, n_files, files_rama_liquid]
//   xxIS : N x 5   [T, v, u, h, s]
//   xxMD : {nom, unitatT(0=C,1=K), mescla, Tc_K, Pc_MPa}
//
// Estat retornat com a llista:
//   {T, P, v, u, h, s, x, regio, avis}
//   regio: 0=liquid comprimit, 1=bifasic, 2=vapor sobreescalfat, -1=error
// ===================================================================

// dades de la substancia activa
EXPORT TST, TSP, TIX, TIS, TMD;

// UNA declaracio per linia i sempre en majuscules.
// PPL admet 'EXPORT A,B,C;' (sense valor) o 'EXPORT A:=v;' (una sola), pero
// NO 'EXPORT A:=1, B:=2;': aixo dona error de sintaxi. I els noms en
// minuscula poden xocar amb variables predefinides de Home.
EXPORT TCUR:=1;       // index de la substancia activa dins TSUBS
EXPORT TERR:="";      // text de l'ultim error

// camps del formulari. TD1/TD2 son l'index de magnitud triat als
// desplegables (1=P 2=T 3=x 4=v 5=u 6=h 7=s) i TV1/TV2 el seu valor.
// Son globals a proposit: aixi la pantalla recorda l'ultima combinacio.
EXPORT TD1:=1;
EXPORT TD2:=6;
EXPORT TV1:=0;
EXPORT TV2:=0;
// Llista FIXA dels desplegables. Porta les DUES unitats de temperatura a
// proposit: INPUT construeix les etiquetes una sola vegada, aixi que si la
// unitat depengues de la substancia no es podria refrescar en canviar-la
// dins del formulari. Oferint totes dues, l'etiqueta no ha de canviar mai i
// es la calculadora qui converteix (TCONV).
//   1 P | 2 T[C] | 3 T[K] | 4 x | 5 v | 6 u | 7 h | 8 s
EXPORT TPROP:={"P [MPa]","T [C]","T [K]","x",
               "v [m3/kg]","u [kJ/kg]","h [kJ/kg]","s [kJ/kgK]"};

// -------------------------------------------------------------------
// utilitats numeriques
// -------------------------------------------------------------------
EXPORT TLIN(zxv, zxa, zxb, zya, zyb)
BEGIN
  IF zxb == zxa THEN RETURN zya; END;
  RETURN zya + (zxv - zxa) * (zyb - zya) / (zxb - zxa);
END;

// Cerca el tram [i, i+1] de la columna c que conte x, dins de les files
// r0..r0+n-1. Retorna 0 si x cau fora del rang tabulat: mai s'extrapola.
EXPORT TBRK(zm, zc, zx, zrbase, zn)
BEGIN
  LOCAL zlo, zhi, zmid;
  IF zn < 2 THEN RETURN 0; END;
  IF zx < zm(zrbase, zc) - 1E-9 THEN RETURN 0; END;
  IF zx > zm(zrbase + zn - 1, zc) + 1E-9 THEN RETURN 0; END;
  zlo := zrbase; zhi := zrbase + zn - 1;
  WHILE zhi - zlo > 1 DO
    zmid := IP((zlo + zhi) / 2);
    IF zm(zmid, zc) <= zx THEN zlo := zmid; ELSE zhi := zmid; END;
  END;
  RETURN zlo;
END;

// -------------------------------------------------------------------
// carrega de substancia
// -------------------------------------------------------------------
EXPORT TLOAD(zii)
BEGIN
  LOCAL zp;
  IF zii < 1 OR zii > SIZE(TSUBS) THEN RETURN 0; END;
  zp := TSUBS(zii);
  TST := EXPR(zp + "ST");
  TSP := EXPR(zp + "SP");
  TIX := EXPR(zp + "IX");
  TIS := EXPR(zp + "IS");
  TMD := EXPR(zp + "MD");
  TCUR := zii;
  RETURN 1;
END;

// Index del desplegable -> id intern de magnitud (1=P 2=T 3=x 4=v 5=u 6=h 7=s).
// Les dues entrades de temperatura cauen totes dues al 2.
EXPORT TIDX(zd)
BEGIN
  IF zd <= 2 THEN RETURN zd; END;
  IF zd == 3 THEN RETURN 2; END;
  RETURN zd - 1;
END;

// Passa el valor a la unitat en que esta tabulada la substancia. La resta de
// magnituds no es toquen: nomes la temperatura canvia d'unitat entre taules.
EXPORT TCONV(zd, zv)
BEGIN
  IF zd == 2 AND TMD(2) == 1 THEN RETURN zv + 273.15; END;
  IF zd == 3 AND TMD(2) == 0 THEN RETURN zv - 273.15; END;
  RETURN zv;
END;

// Temperatura de la taula passada a Kelvin, per ensenyar-la sempre a totes
// dues escales.
EXPORT TTOK(zt)
BEGIN
  IF TMD(2) == 1 THEN RETURN zt; END;
  RETURN zt + 273.15;
END;

// -------------------------------------------------------------------
// corba de saturacio
// -------------------------------------------------------------------
// Retorna {Pl, Pv, vl, vv, ul, uv, hl, hv, sl, sv} interpolat a T.
EXPORT TSATT(zt)
BEGIN
  LOCAL zii, zn, zk, zr, zf, zd;
  zd := DIM(TST); zn := zd(1);
  zii := TBRK(TST, 1, zt, 1, zn);
  IF zii == 0 THEN TERR := "T fora de rang (saturacio)"; RETURN {}; END;
  zf := TLIN(zt, TST(zii,1), TST(zii+1,1), 0, 1);
  zr := {};
  FOR zk FROM 2 TO 11 DO
    zr(zk-1) := TST(zii,zk) + zf * (TST(zii+1,zk) - TST(zii,zk));
  END;
  RETURN zr;
END;

// Retorna {Tl, Tv, vl, vv, ul, uv, hl, hv, sl, sv} interpolat a P.
// S'usa aquesta taula, i no la de T, sempre que la dada sigui P:
// reconstruir T_sat(P) des de la taula per T dona fins a 0,3 C d'error
// perque P_sat(T) es exponencial.
EXPORT TSATP(zp)
BEGIN
  LOCAL zii, zn, zk, zr, zf, zd;
  zd := DIM(TSP); zn := zd(1);
  zii := TBRK(TSP, 1, zp, 1, zn);
  IF zii == 0 THEN TERR := "P fora de rang (saturacio)"; RETURN {}; END;
  zf := TLIN(zp, TSP(zii,1), TSP(zii+1,1), 0, 1);
  zr := {};
  FOR zk FROM 2 TO 11 DO
    zr(zk-1) := TSP(zii,zk) + zf * (TSP(zii+1,zk) - TSP(zii,zk));
  END;
  RETURN zr;
END;

// -------------------------------------------------------------------
// isobares
// -------------------------------------------------------------------
// Interpola dins de la isobara k a la temperatura T, a la branca ph
// (0 = liquid, 1 = vapor). Retorna {v,u,h,s} o {} si T no hi cau.
// Cada isobara porta liquid subrefredat, el salt de saturacio i vapor
// sobreescalfat; TIX(k,4) diu quantes files son de la branca liquida.
EXPORT TISOT(zk, zt, zph)
BEGIN
  LOCAL zrbase, znn, znl, zii, zf, zres;
  LOCAL zj;
  zrbase := TIX(zk,2); znn := TIX(zk,3); znl := TIX(zk,4);
  // fase: 0 = liquid, 1 = vapor, 2 = tota la isobara (per sobre del punt
  // critic no hi ha salt de fase i la isobara es una sola branca)
  IF zph == 0 THEN
    IF znl < 2 THEN RETURN {}; END;
    zii := TBRK(TIS, 1, zt, zrbase, znl);
  END;
  IF zph == 1 THEN
    IF znn - znl < 2 THEN RETURN {}; END;
    zii := TBRK(TIS, 1, zt, zrbase + znl, znn - znl);
  END;
  IF zph == 2 THEN
    IF znn < 2 THEN RETURN {}; END;
    zii := TBRK(TIS, 1, zt, zrbase, znn);
  END;
  IF zii == 0 THEN RETURN {}; END;
  zf := TLIN(zt, TIS(zii,1), TIS(zii+1,1), 0, 1);
  zres := {};
  FOR zj FROM 2 TO 5 DO
    zres(zj-1) := TIS(zii,zj) + zf * (TIS(zii+1,zj) - TIS(zii,zj));
  END;
  RETURN zres;
END;

// Llegeix una isobara sense exigir fase. S'usa per sobre del punt critic.
// Si la isobara SI te salt (una veina subcritica), s'agafa la branca que
// conte aquesta T, que es l'unica amb sentit fisic.
EXPORT TISOA(zk, zt)
BEGIN
  LOCAL zr;
  zr := TISOT(zk, zt, 0);
  IF SIZE(zr) == 4 THEN RETURN zr; END;
  zr := TISOT(zk, zt, 1);
  IF SIZE(zr) == 4 THEN RETURN zr; END;
  RETURN TISOT(zk, zt, 2);
END;

// Doble interpolacio: primer en T dins de cada isobara, despres en P.
// Aquest ordre es el que es fa a ma a classe, i es respecta perque el
// resultat coincideixi amb la solucio oficial.
//
// Cas espinos: a la temperatura T, una de les dues isobares veines pot
// estar a l'ALTRA fase (a P1 < P la saturacio arriba abans, aixi que un
// liquid comprimit a P pot ser vapor a P1). Interpolar entre un liquid i
// un vapor donaria un numero sense sentit. Quan passa, s'agafa el propi
// estat de saturacio a la mateixa T com a punt de suport i s'avisa.
EXPORT TSPH(zp, zt, zph)
BEGIN
  LOCAL zm, zk, za, zb, zra, zrb;
  LOCAL zres, zj, zsa, zpsat, zrsat, zd;
  LOCAL zhit;
  zd := DIM(TIX); zm := zd(1);
  TERR := "";
  // Isobara exactament tabulada. Es marca amb una bandera i es surt amb
  // BREAK: PPL no admet RETURN des de dins d'un FOR.
  zhit := 0;
  FOR zk FROM 1 TO zm DO
    IF ABS(TIX(zk,1) - zp) < 1E-12 THEN
      zhit := zk;
      BREAK;
    END;
  END;
  IF zhit > 0 THEN
    zres := TISOT(zhit, zt, zph);
    IF SIZE(zres) == 0 THEN TERR := "T fora de rang a aquesta P"; END;
    RETURN zres;
  END;
  zk := TBRK(TIX, 1, zp, 1, zm);
  IF zk == 0 THEN TERR := "P fora de rang (isobares)"; RETURN {}; END;
  za := TIX(zk,1); zb := TIX(zk+1,1);
  IF zph == 2 THEN
    zra := TISOA(zk, zt); zrb := TISOA(zk+1, zt);
  ELSE
    zra := TISOT(zk, zt, zph); zrb := TISOT(zk+1, zt, zph);
  END;

  IF SIZE(zra) == 4 AND SIZE(zrb) == 4 THEN
    zres := {};
    // Lineal per a les quatre, v inclosa: la app ha de donar EXACTAMENT
    // el mateix que surt interpolant a ma a l'examen. No es el mes exacte
    // (v ~ ZRT/P es gairebe hiperbolica, i on el PDF fa un salt gran la
    // recta se'n va), pero es l'unic numero que es pot justificar al paper.
    // NO tornar a posar 1/P aqui: ho vigila test_engine.py.
    FOR zj FROM 1 TO 4 DO
      zres(zj) := TLIN(zp, za, zb, zra(zj), zrb(zj));
    END;
    RETURN zres;
  END;

  // una veina no te aquesta fase a aquesta T -> frontera de saturacio
  zsa := TSATT(zt);
  IF SIZE(zsa) == 0 THEN RETURN {}; END;
  TERR := "veina en altra fase: s'usa saturacio";
  IF zph == 0 THEN
    zpsat := zsa(1);
    zrsat := {zsa(3), zsa(5), zsa(7), zsa(9)};
  ELSE
    zpsat := zsa(2);
    zrsat := {zsa(4), zsa(6), zsa(8), zsa(10)};
  END;
  IF SIZE(zra) == 4 THEN
    zres := {};
    FOR zj FROM 1 TO 4 DO zres(zj) := TLIN(zp, za, zpsat, zra(zj), zrsat(zj)); END;
    RETURN zres;
  END;
  IF SIZE(zrb) == 4 THEN
    zres := {};
    FOR zj FROM 1 TO 4 DO zres(zj) := TLIN(zp, zb, zpsat, zrb(zj), zrsat(zj)); END;
    RETURN zres;
  END;
  TERR := "cap isobara veina te aquesta fase a aquesta T";
  RETURN {};
END;

// -------------------------------------------------------------------
// estats
// -------------------------------------------------------------------
// (P, T)
EXPORT TPT(zp, zt)
BEGIN
  LOCAL zsa, zreg, zr, zav;
  TERR := "";
  // Per sobre de la pressio critica no hi ha corba de saturacio: no hi ha res
  // a comparar i s'ha d'anar directe a la isobara. Regio 3 = supercritic.
  // Compte: amb P < Pc i T > Tc l'estat es vapor sobreescalfat corrent, i
  // aquest cas el resol be el cami normal.
  IF TMD(5) > 0 AND zp > TMD(5) THEN
    zr := TSPH(zp, zt, 2);
    IF SIZE(zr) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
    RETURN {zt, zp, zr(1), zr(2), zr(3), zr(4), -1, 3, TERR};
  END;
  zsa := TSATP(zp);
  IF SIZE(zsa) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  IF zt < zsa(1) - 1E-9 THEN zreg := 0;
  ELSE IF zt > zsa(2) + 1E-9 THEN zreg := 2;
  ELSE
    // just a la saturacio: (P,T) no determina l'estat, cal x
    RETURN {zt, zp, zsa(3), zsa(5), zsa(7), zsa(9), 0, 1,
            "(P,T) satura: x indeterminat, s'agafa x=0"};
  END; END;
  zr := TSPH(zp, zt, IFTE(zreg == 0, 0, 1));
  zav := TERR;
  IF SIZE(zr) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  RETURN {zt, zp, zr(1), zr(2), zr(3), zr(4), -1, zreg, zav};
END;

// (P, x)
EXPORT TPX(zp, zx)
BEGIN
  LOCAL zsa, zt;
  TERR := "";
  IF zx < 0 OR zx > 1 THEN RETURN {0,0,0,0,0,0,0,-1,"x ha d'estar entre 0 i 1"}; END;
  zsa := TSATP(zp);
  IF SIZE(zsa) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  zt := zsa(1) + zx * (zsa(2) - zsa(1));
  RETURN {zt, zp,
          zsa(3) + zx*(zsa(4)-zsa(3)), zsa(5) + zx*(zsa(6)-zsa(5)),
          zsa(7) + zx*(zsa(8)-zsa(7)), zsa(9) + zx*(zsa(10)-zsa(9)),
          zx, 1, IFTE(TMD(3) == 1, "mescla: T bombolla/rosada", "")};
END;

// (T, x)
EXPORT TTX(zt, zx)
BEGIN
  LOCAL zsa, zp;
  TERR := "";
  IF zx < 0 OR zx > 1 THEN RETURN {0,0,0,0,0,0,0,-1,"x ha d'estar entre 0 i 1"}; END;
  zsa := TSATT(zt);
  IF SIZE(zsa) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  zp := zsa(1) + zx * (zsa(2) - zsa(1));
  RETURN {zt, zp,
          zsa(3) + zx*(zsa(4)-zsa(3)), zsa(5) + zx*(zsa(6)-zsa(5)),
          zsa(7) + zx*(zsa(8)-zsa(7)), zsa(9) + zx*(zsa(10)-zsa(9)),
          zx, 1, ""};
END;

// (P, propietat) amb pr = 1:v  2:u  3:h  4:s
// Encaminament: es compara y amb y_f(P) i y_g(P) de la taula de saturacio.
//   y < y_f          -> liquid comprimit   -> inversa dins de la isobara
//   y_f <= y <= y_g  -> bifasic            -> x = (y-y_f)/(y_g-y_f), directe
//   y > y_g          -> vapor sobreescalfat-> inversa dins de la isobara
EXPORT TPY(zp, zpr, zy)
BEGIN
  LOCAL zsa, zyf, zyg, zx, zt, zr;
  LOCAL zreg, zav;
  TERR := "";
  // Per sobre de la pressio critica no hi ha corba de saturacio: no hi ha
  // y_f ni y_g amb que decidir la regio, i TSATP torna {}. S'inverteix
  // directament dins la isobara, que alli es una rama continua. Regio 3 =
  // supercritic, com a TPT i TTY.
  IF TMD(5) > 0 AND zp > TMD(5) THEN
    zt := TINVP(zp, zpr, zy, 3);
    IF zt == -99999 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
    zr := TSPH(zp, zt, 2);
    zav := TERR;
    IF SIZE(zr) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
    RETURN {zt, zp, zr(1), zr(2), zr(3), zr(4), -1, 3, zav};
  END;
  zsa := TSATP(zp);
  IF SIZE(zsa) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  zyf := zsa(1 + 2*zpr); zyg := zsa(2 + 2*zpr);
  IF zy >= zyf AND zy <= zyg THEN
    IF zyg == zyf THEN zx := 0; ELSE zx := (zy - zyf) / (zyg - zyf); END;
    RETURN TPX(zp, zx);
  END;
  IF zy < zyf THEN zreg := 0; ELSE zreg := 2; END;
  zt := TINVP(zp, zpr, zy, zreg);
  IF TERR <> "" AND zt == -99999 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  zr := TSPH(zp, zt, IFTE(zreg == 0, 0, 1));
  zav := TERR;
  IF SIZE(zr) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
  RETURN {zt, zp, zr(1), zr(2), zr(3), zr(4), -1, zreg, zav};
END;

// Cerca inversa: donat (P, y) troba T a la regio monofasica.
// S'inverteix la MATEIXA funcio que fa el calcul directe (doble
// interpolacio) per biseccio, aixi anada i tornada son coherents.
// La monotonia de h i s amb T a P constant garanteix solucio unica.
EXPORT TINVP(zp, zpr, zy, zreg)
BEGIN
  LOCAL zm, zk, zkaa, zkbb, zkk, zph;
  LOCAL zlo, zhi, zii, zrr, zflo, zfhi;
  LOCAL zmid, zfm, zit, zd;
  // zreg: 0 liquid, 2 vapor, 3 supercritic -> fase 0, 1 o 2
  zph := 1;
  IF zreg == 0 THEN zph := 0; END;
  IF zreg == 2 THEN zph := 2; END;
  // Per sobre de Pc la isobara es UNA rama continua i l'index la desa amb
  // TIX(k,4) == TIX(k,3): no hi ha rama vapor separada, aixi que demanar-la
  // deixava l'interval de cerca buit. Es llegeix sense exigir fase, igual
  // que fa TTY amb zfa := 2.
  IF zreg == 3 THEN zph := 2; END;
  zd := DIM(TIX); zm := zd(1);
  // Se acota con LAS MISMAS isobaras que usara el calculo directo (las que
  // encierran P, o la exacta si esta tabulada). Usar mas isobaras estrecharia
  // el intervalo de busqueda de forma artificial y podria dejar fuera la
  // solucion.
  zkaa := 0;
  FOR zk FROM 1 TO zm DO
    IF ABS(TIX(zk,1) - zp) < 1E-12 THEN zkaa := zk; zkbb := zk; END;
  END;
  IF zkaa == 0 THEN
    zkaa := TBRK(TIX, 1, zp, 1, zm);
    IF zkaa == 0 THEN TERR := "P fora de rang (isobares)"; RETURN -99999; END;
    zkbb := zkaa + 1;
  END;
  zlo := -1E12; zhi := 1E12;
  FOR zkk FROM zkaa TO zkbb DO
    zii := TIX(zkk,2);
    IF zph == 0 THEN
      IF TIX(zkk,4) >= 2 THEN
        zlo := MAX(zlo, TIS(zii,1));
        zhi := MIN(zhi, TIS(zii + TIX(zkk,4) - 1, 1));
      END;
    END;
    IF zph == 1 THEN
      IF TIX(zkk,3) - TIX(zkk,4) >= 2 THEN
        zlo := MAX(zlo, TIS(zii + TIX(zkk,4), 1));
        zhi := MIN(zhi, TIS(zii + TIX(zkk,3) - 1, 1));
      END;
    END;
    IF zph == 2 THEN
      IF TIX(zkk,3) >= 2 THEN
        zlo := MAX(zlo, TIS(zii,1));
        zhi := MIN(zhi, TIS(zii + TIX(zkk,3) - 1, 1));
      END;
    END;
  END;
  IF zlo >= zhi THEN TERR := "sense dades d'aquesta fase a aquesta P"; RETURN -99999; END;
  zrr := TSPH(zp, zlo, zph); IF SIZE(zrr) == 0 THEN TERR := "inversa: extrem invalid"; RETURN -99999; END;
  zflo := zrr(zpr) - zy;
  zrr := TSPH(zp, zhi, zph); IF SIZE(zrr) == 0 THEN TERR := "inversa: extrem invalid"; RETURN -99999; END;
  zfhi := zrr(zpr) - zy;
  IF zflo * zfhi > 0 THEN
    TERR := "valor fora del rang tabulat a aquesta P";
    RETURN -99999;
  END;
  FOR zit FROM 1 TO 60 DO
    zmid := (zlo + zhi) / 2;
    zrr := TSPH(zp, zmid, zph);
    IF SIZE(zrr) == 0 THEN BREAK; END;
    zfm := zrr(zpr) - zy;
    IF zflo * zfm <= 0 THEN zhi := zmid; zfhi := zfm; ELSE zlo := zmid; zflo := zfm; END;
    IF zhi - zlo < 1E-9 THEN BREAK; END;
  END;
  TERR := "";
  RETURN (zlo + zhi) / 2;
END;

// (T, propietat)
EXPORT TTY(zt, zpr, zy)
BEGIN
  LOCAL zsa, zyf, zyg, zx, zp, zr;
  LOCAL zreg, zav, zm, zk, zylo, zyhi;
  LOCAL zplo, zphi, zrr, zgot, zd, zout;
  LOCAL zfa;
  TERR := "";
  // Per sobre de la temperatura critica la taula de saturacio s'acaba: no hi
  // ha y_f ni y_g amb que decidir la regio. S'inverteix directament sobre les
  // isobares i la regio la marca despres la pressio que surti.
  zreg := -9;
  IF TMD(4) > 0 AND TTOK(zt) > TMD(4) THEN zreg := 3; END;
  IF zreg == -9 THEN
    zsa := TSATT(zt);
    IF SIZE(zsa) == 0 THEN RETURN {0,0,0,0,0,0,0,-1,TERR}; END;
    zyf := zsa(1 + 2*zpr); zyg := zsa(2 + 2*zpr);
    IF zy >= zyf AND zy <= zyg THEN
      IF zyg == zyf THEN zx := 0; ELSE zx := (zy - zyf) / (zyg - zyf); END;
      RETURN TTX(zt, zx);
    END;
    IF zy < zyf THEN zreg := 0; ELSE zreg := 2; END;
  END;
  // fase amb que es llegeix cada isobara: 0 liquid, 1 vapor, 2 sense exigir
  zfa := 1;
  IF zreg == 0 THEN zfa := 0; END;
  IF zreg == 3 THEN zfa := 2; END;
  // es recorren les isobares que tenen aquesta fase a aquesta T i es
  // busca el tram de P on y canvia de banda
  // El resultat s'acumula a zout i es retorna DESPRES del bucle: PPL no
  // admet RETURN des de dins d'un FOR.
  zd := DIM(TIX); zm := zd(1); zgot := 0; zplo := 0; zylo := 0;
  zout := {};
  FOR zk FROM 1 TO zm DO
    IF zfa == 2 THEN zrr := TISOA(zk, zt); ELSE zrr := TISOT(zk, zt, zfa); END;
    IF SIZE(zrr) == 4 THEN
      zphi := TIX(zk,1); zyhi := zrr(zpr);
      IF zgot == 1 THEN
        IF (zylo - zy) * (zyhi - zy) <= 0 THEN
          zp := TLIN(zy, zylo, zyhi, zplo, zphi);
          zr := TSPH(zp, zt, zfa);
          zav := TERR;
          IF SIZE(zr) == 0 THEN
            zout := {0,0,0,0,0,0,0,-1,TERR};
          ELSE
            // per sobre de Tc pero amb P per sota de Pc l'estat es vapor
            // sobreescalfat corrent, no supercritic
            IF zreg == 3 AND TMD(5) > 0 AND zp <= TMD(5) THEN zreg := 2; END;
            zout := {zt, zp, zr(1), zr(2), zr(3), zr(4), -1, zreg, zav};
          END;
          BREAK;
        END;
      END;
      zplo := zphi; zylo := zyhi; zgot := 1;
    END;
  END;
  IF SIZE(zout) == 9 THEN RETURN zout; END;
  RETURN {0,0,0,0,0,0,0,-1,"valor fora del rang tabulat a aquesta T"};
END;
